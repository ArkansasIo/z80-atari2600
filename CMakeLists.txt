cmake_minimum_required(VERSION 3.20)
project(z80_atari2600_toolkit VERSION 0.1.0 LANGUAGES CXX)
option(TOOLKIT_BUILD_TESTS "Build unit tests" ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(z80core src/z80/cpu.cpp)
target_include_directories(z80core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(z80core PUBLIC cxx_std_23)
if(MSVC)
 target_compile_options(z80core PRIVATE /W4 /permissive-)
else()
 target_compile_options(z80core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(z80-runner src/main.cpp)
target_link_libraries(z80-runner PRIVATE z80core)
if(TOOLKIT_BUILD_TESTS)
 enable_testing()
 add_executable(z80-tests tests/test_cpu.cpp)
 target_link_libraries(z80-tests PRIVATE z80core)
 add_test(NAME z80-tests COMMAND z80-tests)
endif()
